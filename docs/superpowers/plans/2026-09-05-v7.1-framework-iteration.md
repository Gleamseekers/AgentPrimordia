```go
type ApprovalGate interface {

    WaitApproval(ctx context.Context, action string) error
}
```

```go
// 告警严重度
type AlertSeverity string

const (
    SeverityInfo     AlertSeverity = "info"
    SeverityWarning  AlertSeverity = "warning"
    SeverityCritical AlertSeverity = "critical"
)

// AlertRule 告警规则接口
type AlertRule interface {
    Name() string
    Evaluate(store *CorrelationStore) ([]AlertEvent, error)
}

// AlertEvent 告警事件
type AlertEvent struct {
    Rule      string        `json:"rule"`
    Severity  AlertSeverity `json:"severity"`
    Message   string        `json:"message"`
    TraceIDs  []string      `json:"trace_ids,omitempty"`
    Threshold float64       `json:"threshold"`
    Actual    float64       `json:"actual"`
}

// thresholdRule 阈值告警规则（内部实现用于测试）
type thresholdRule struct {
    name      string
    threshold float64
    severity  AlertSeverity
    metricFn  func(store *CorrelationStore) float64
}

func (r *thresholdRule) Name() string { return r.name }
func (r *thresholdRule) Evaluate(store *CorrelationStore) ([]AlertEvent, error) {
    val := r.metricFn(store)
    if val > r.threshold {
        return []AlertEvent{{
            Rule:      r.name,
            Severity:  r.severity,
            Message:   r.name + " exceeded threshold",
            Threshold: r.threshold,
            Actual:    val,
        }}, nil
    }
    return nil, nil
}
```

```go
// internal/tools/intelligence/intelligence.go
package intelligence

import (
    "context"
    "sync"
    "time"
)

// ToolIntelligence 统一工具智能入口
type ToolIntelligence struct {
    mu       sync.Mutex
    detector GapDetector
    creator  ToolCreator
    profiler ToolProfiler
    tuner    ToolTuner
    selector ToolSelector
    catalog  *ToolCatalog
    matcher  TaskMatcher
}

// GapDetector 缺口检测器——从失败轨迹识别 missing_tool 模式
type GapDetector interface {
    Detect(ctx context.Context, trace []ToolCallRecord) ([]GapCandidate, error)
}

// ToolCreator 工具生成器——从缺口生成新工具
type ToolCreator interface {
    Create(ctx context.Context, gap GapCandidate) (*ToolArtifact, error)
}

// ToolProfiler 工具性能画像——per-tool 统计
type ToolProfiler interface {
    Record(ctx context.Context, usage ToolUsageRecord) error
    Profile(ctx context.Context, toolName string) (*ToolProfile, error)
    AllProfiles(ctx context.Context) (map[string]*ToolProfile, error)
}

// ToolTuner 参数调优器——基于性能画像数据驱动调优
type ToolTuner interface {
    SuggestTuning(ctx context.Context, toolName string, profile *ToolProfile) (*TuningSuggestion, error)
    ApplyTuning(ctx context.Context, toolName string, suggestion *TuningSuggestion) error
}

// ToolSelector 工具选择优化——学习最优选择策略
type ToolSelector interface {
    Select(ctx context.Context, task string, candidates []string) (string, error)
    RecordOutcome(ctx context.Context, toolName string, success bool) error
}

// 数据类型

type GapCandidate struct {
    Kind        string    `json:"kind"` // missing_tool / repeated_failure
    Key         string    `json:"key"`
    Count       int       `json:"count"`
    SampleError string    `json:"sample_error,omitempty"`
    FirstSeen   time.Time `json:"first_seen"`
    LastSeen    time.Time `json:"last_seen"`
}

type ToolArtifact struct {
    ID          string `json:"id"`
    Name        string `json:"name"`
    Description string `json:"description"`
    ArtifactSHA string `json:"artifact_sha"`
    Artifact    []byte `json:"artifact"`
}

type ToolCallRecord struct {
    ToolName  string        `json:"tool_name"`
    Args      string        `json:"args"`
    Result    string        `json:"result"`
    Error     string        `json:"error,omitempty"`
    Duration  time.Duration `json:"duration"`
    Success   bool          `json:"success"`
    Timestamp time.Time     `json:"timestamp"`
}

type ToolUsageRecord struct {
    ToolName string
    Success  bool
    Duration time.Duration
    Tokens   int
}

type ToolProfile struct {
    ToolName     string        `json:"tool_name"`
    TotalCalls   int           `json:"total_calls"`
    SuccessRate  float64       `json:"success_rate"`
    AvgDuration  time.Duration `json:"avg_duration"`
    P95Duration  time.Duration `json:"p95_duration"`
    AvgTokens    int           `json:"avg_tokens"`
    LastUsed     time.Time     `json:"last_used"`
}

type TuningSuggestion struct {
    ToolName    string            `json:"tool_name"`
    Parameter   string            `json:"parameter"`
    CurrentVal  string            `json:"current_val"`
    SuggestedVal string           `json:"suggested_val"`
    Confidence  float64           `json:"confidence"`
    Reason      string            `json:"reason"`
}

// NewToolIntelligence 构造器
func NewToolIntelligence(detector GapDetector, creator ToolCreator, profiler ToolProfiler, tuner ToolTuner, selector ToolSelector) *ToolIntelligence {
    return &ToolIntelligence{
        detector: detector,
        creator:  creator,
        profiler: profiler,
        tuner:    tuner,
        selector: selector,

    }
}
```

```go
    data, err := os.ReadFile(path)

func loadDone(path string) map[string]bool {
    done := make(map[string]bool)
    f, err := os.Open(path)
    if err != nil {
        return done
    }
    defer f.Close()
    dec := json.NewDecoder(f)
    for {
        var r unitResult
        if err := dec.Decode(&r); err != nil {
            break
        }
        if r.Error == "" {
            done[r.Item+"/"+r.Arm] = true
        }
    }
    return done
}

func runUnit(ctx context.Context, provider *llm.OpenAIProvider, task mmTask, arm string) unitResult {
    start := time.Now()

    // 创建沙箱
    sandbox, err := os.MkdirTemp("", "mm-"+task.ID+"-")
    if err != nil {
        return unitResult{Item: task.ID, Arm: arm, Error: err.Error()}
    }
    defer os.RemoveAll(sandbox)

    // 注入 fixtures
    for _, fix := range task.Fixtures {
        fullPath := filepath.Join(sandbox, fix.Path)
        os.MkdirAll(filepath.Dir(fullPath), 0o755)
        content := fix.Inline
        if content == "" {
            content = fix.InlineB64
        }
        os.WriteFile(fullPath, []byte(content), 0o644)
    }

    // 构建工具包
    reg := tools.NewRegistry()
    fs := builtin.NewFileSystem(sandbox)
    sh := builtin.NewShell(builtin.WithWorkDir(sandbox))
    reg.Register(fs)
    reg.Register(sh)

    // 检查点
    ckpt, _ := persist.NewSQLiteCheckpointStore(filepath.Join(sandbox, "checkpoint.db"))

    // A 臂：纯文本（图像信息通过 OCR 文本注入 system prompt）
    // B 臂：多模态（图像通过 ContentPart 传入）
    sysPrompt := "你是一个文件处理助手。工作目录: " + sandbox
    if arm == "A" {
        sysPrompt += "\n注意：图像文件无法直接查看，请使用 shell 工具尝试提取文本信息。"
    }

    ag := agent.NewAgent("mm-agent", sysPrompt, provider,
        agent.WithMaxTurns(15),
        agent.WithToolkit(reg),
        agent.WithCheckpointStore(ckpt),
        agent.WithSessionID(task.ID+"-"+arm),
    )

    var msg agent.Message
    if arm == "B" {
        // B 臂：构建多模态消息
        parts := []multimodal.ContentPart{{Type: "text", Text: task.Task}}
        for _, fix := range task.Fixtures {
            if strings.HasPrefix(fix.Path, "images/") || strings.HasPrefix(fix.Path, "audio/") {
                parts = append(parts, multimodal.ContentPart{
                    Type: "image_url",
                    Data: fix.InlineB64,
                    MIME: "image/png",
                })
            }
        }
        msg = agent.Message{Role: "user", Content: task.Task, ContentParts: parts}
    } else {
        msg = agent.UserMessage(task.Task)
    }

    resp, err := ag.Run(ctx, msg)
    if err != nil {
        return unitResult{Item: task.ID, Arm: arm, Error: err.Error(), DurationSec: int(time.Since(start).Seconds())}
    }

    // 判定：检查 result.txt 或响应中是否包含 expected
    success := false
    resultContent, _ := os.ReadFile(filepath.Join(sandbox, "result.txt"))
    combined := string(resultContent) + " " + resp.Content
    if strings.Contains(strings.ToLower(combined), strings.ToLower(task.Expected)) {
        success = true
    }

    return unitResult{
        Item:        task.ID,
        Arm:         arm,
        Success:     success,
        Turns:       resp.Turns,
        DurationSec: int(time.Since(start).Seconds()),
    }
}
```

```go
// internal/agent/planning/plan_state.go
package planning

import (
    "fmt"
    "time"
)

// Transition 状态转换
func (mp *ManagedPlan) Transition(to PlanState, reason string) error {
    // 终态不可转换
    if mp.State == PlanStateCompleted || mp.State == PlanStateFailed {
        return fmt.Errorf("plan in terminal state %s, cannot transition to %s", mp.State, to)
    }

    // 合法转换表
    valid := map[PlanState][]PlanState{
        PlanStatePending: {PlanStateActive, PlanStateFailed},
        PlanStateActive:  {PlanStateBlocked, PlanStateCompleted, PlanStateFailed},
        PlanStateBlocked: {PlanStateActive, PlanStateFailed},
    }

    allowed := false
    for _, s := range valid[mp.State] {
        if s == to {
            allowed = true
            break
        }
    }
    if !allowed {
        return fmt.Errorf("invalid transition %s → %s", mp.State, to)
    }

    mp.History = append(mp.History, PlanTransition{
        From: mp.State,
        To:   to,
        At:   time.Now(),
        Why:  reason,
    })
    mp.State = to
    mp.UpdatedAt = time.Now()
    return nil
}
```

```go
func (r *LLMReplanner) ShouldReplan(ctx context.Context, plan *Plan, observation string) (bool, string) {
    // 构建状态摘要
    var statusLines []string
    for _, st := range plan.SubTasks {
        statusLines = append(statusLines, fmt.Sprintf("  %s: %s", st.ID, st.Status))
    }
    statusSummary := strings.Join(statusLines, "\n")

    prompt := fmt.Sprintf(`你是一个任务规划器。当前计划状态：
目标: %s
子任务状态:
%s

最新观察: %s

请判断是否需要重新规划。只回答"是，<原因>"或"否，<原因>"。`, plan.Goal, statusSummary, observation)

    resp, err := r.provider.Complete(ctx, &llm.CompletionRequest{
        Messages:    []llm.ChatMessage{{Role: "user", Content: prompt}},
        MaxTokens:   100,
        Temperature: 0,
    })
    if err != nil {
        return false, "replanner error: " + err.Error()
    }

    content := strings.TrimSpace(resp.Content)
    if strings.HasPrefix(content, "是") {
        return true, content
    }
    return false, content
}

func (r *LLMReplanner) Replan(ctx context.Context, plan *Plan, reason string) (*Plan, error) {
    var statusLines []string

%s
失败原因: %s

请输出新的子任务分解，JSON 数组格式：[{"id":"s1","description":"...","depends_on":[]}]`,
        plan.Goal, strings.Join(statusLines, "\n"), reason)

    resp, err := r.provider.Complete(ctx, &llm.CompletionRequest{
        Messages:    []llm.ChatMessage{{Role: "user", Content: prompt}},
        MaxTokens:   500,
```

```go
package planning

import (
    "context"
    "fmt"
    "strings"

    "github.com/Gleamseekers/AgentPrimordia/agentprimordia/internal/llm"
)

// DeadlockDetector 死路检测器——连续 N 个子任务失败视为死锁
type DeadlockDetector struct {
    threshold int
}

func NewDeadlockDetector(threshold int) *DeadlockDetector {
    if threshold < 2 {
        threshold = 2
    }
    return &DeadlockDetector{threshold: threshold}
}

func (d *DeadlockDetector) DetectDeadlock(ctx context.Context, plan *Plan, failedSubTask string) bool {
    // 检查从末尾往前是否有连续 threshold 个失败
    consecutive := 0
    for i := len(plan.SubTasks) - 1; i >= 0; i-- {
        if plan.SubTasks[i].Status == TaskFailed {
            consecutive++
        } else {
            break
        }
    }
    return consecutive >= d.threshold
}

// LLMRecoveryStrategy 基于 LLM 的恢复策略
type LLMRecoveryStrategy struct {
    provider llm.Provider
}

func NewLLMRecoveryStrategy(provider llm.Provider) *LLMRecoveryStrategy {
    return &LLMRecoveryStrategy{provider: provider}
}

func (r *LLMRecoveryStrategy) Recover(ctx context.Context, plan *Plan, failedSubTask string) (*Plan, error) {
    var context_lines []string
    for _, st := range plan.SubTasks {
        context_lines = append(context_lines, fmt.Sprintf("  %s [%s]: %s", st.ID, st.Status, st.Description))
    }

    prompt := fmt.Sprintf(`任务执行陷入死路，需要替代方案。
目标: %s
当前计划:
%s
失败子任务: %s

请输出替代子任务，JSON 数组格式：[{"id":"...","description":"...","depends_on":[...]}]`,
        plan.Goal, strings.Join(context_lines, "\n"), failedSubTask)

    resp, err := r.provider.Complete(ctx, &llm.CompletionRequest{
        Messages:    []llm.ChatMessage{{Role: "user", Content: prompt}},
        MaxTokens:   300,
        Temperature: 0,
    })
    if err != nil {
        return nil, err
    }

    subTasks, err := parseSubTasks(resp.Content)
    if err != nil {
        return nil, err
    }
    return &Plan{Goal: plan.Goal, SubTasks: subTasks}, nil
}
```

```go
// internal/agent/planning/approval.go
package planning

import (
    "context"
    "fmt"
    "sync"
)

// PolicyApprovalGate 基于策略列表的审批门
type PolicyApprovalGate struct {
    mu              sync.RWMutex
    requiredActions map[string]bool
    pendingApproval map[string]chan struct{}
}

func NewPolicyApprovalGate(actions []string) *PolicyApprovalGate {
    m := make(map[string]bool)
    for _, a := range actions {
        m[a] = true
    }
    return &PolicyApprovalGate{
        requiredActions: m,
        pendingApproval: make(map[string]chan struct{}),
    }
}

func (g *PolicyApprovalGate) RequiresApproval(ctx context.Context, action string) bool {
    g.mu.RLock()
    defer g.mu.RUnlock()
    return g.requiredActions[action]
}

func (g *PolicyApprovalGate) RequestApproval(ctx context.Context, action, reason string) error {
    g.mu.Lock()
    ch := make(chan struct{})
    g.pendingApproval[action] = ch
    g.mu.Unlock()
    // 实际部署中这里会发送通知给人类审批者
    // 当前实现：直接通过（测试模式）
    close(ch)
    return nil
}

func (g *PolicyApprovalGate) WaitApproval(ctx context.Context, action string) error {
    g.mu.RLock()
    ch, ok := g.pendingApproval[action]
    g.mu.RUnlock()
    if !ok {
        return nil // 无需审批
    }
    select {
    case <-ch:
        g.mu.Lock()
        delete(g.pendingApproval, action)
        g.mu.Unlock()
        return nil
    case <-ctx.Done():
        return fmt.Errorf("approval wait cancelled: %w", ctx.Err())
    }
}
```

```go
// internal/observability/alert.go
package observability

import "sync"

// AlertEngine 告警引擎——定期评估规则并产生告警
type AlertEngine struct {
    mu    sync.RWMutex
    store *CorrelationStore
    rules []AlertRule
}

func NewAlertEngine(store *CorrelationStore) *AlertEngine {
    return &AlertEngine{store: store}
}

func (e *AlertEngine) RegisterRule(rule AlertRule) {
    e.mu.Lock()
    defer e.mu.Unlock()
    e.rules = append(e.rules, rule)
}

func (e *AlertEngine) Evaluate() []AlertEvent {
    e.mu.RLock()
    rules := make([]AlertRule, len(e.rules))
    copy(rules, e.rules)
    e.mu.RUnlock()

    var alerts []AlertEvent
    for _, rule := range rules {
        events, err := rule.Evaluate(e.store)
        if err != nil {
            continue
        }
        alerts = append(alerts, events...)
    }
    return alerts
}
```

```go
    mux.Handle("/traces", Handler(store))
```

```go
    c.mu.RLock()
    defer c.mu.RUnlock()
    e, ok := c.tools[id]
    return e, ok
}

    c.mu.RLock()
    defer c.mu.RUnlock()
    result := make([]ToolEntry, 0, len(c.tools))
    for _, e := range c.tools {
        result = append(result, e)
    }
    return result
```

```go
import "strings"

type TaskMatcher struct{}

func NewTaskMatcher() *TaskMatcher {
    return &TaskMatcher{}
}

// Match 基于关键词重叠的任务-工具匹配
func (m *TaskMatcher) Match(task string, tools []ToolEntry) ToolEntry {
    taskLower := strings.ToLower(task)
    var best ToolEntry
    bestScore := 0

    for _, tool := range tools {
        descLower := strings.ToLower(tool.Description)
        // 简单的关键词重叠计分
        score := 0
        for _, word := range strings.Fields(descLower) {
            if len(word) > 1 && strings.Contains(taskLower, word) {
                score++
            }
        }
        if score > bestScore {
            bestScore = score
            best = tool
        }
    }

    if bestScore == 0 {
        return ToolEntry{} // 无匹配
    }
    return best
}
```

```go
// internal/tools/intelligence/optimize/profiler.go
package optimize

import (
    "context"
    "sort"
    "sync"
    "time"
    "github.com/Gleamseekers/AgentPrimordia/agentprimordia/internal/tools/intelligence"
)

type InMemoryProfiler struct {
    mu      sync.Mutex
    records map[string][]intelligence.ToolUsageRecord
}

func NewInMemoryProfiler() *InMemoryProfiler {
    return &InMemoryProfiler{records: make(map[string][]intelligence.ToolUsageRecord)}
}

func (p *InMemoryProfiler) Record(ctx context.Context, usage intelligence.ToolUsageRecord) error {
    p.mu.Lock()
    defer p.mu.Unlock()
    p.records[usage.ToolName] = append(p.records[usage.ToolName], usage)
    return nil
}

func (p *InMemoryProfiler) Profile(ctx context.Context, toolName string) (*intelligence.ToolProfile, error) {
    p.mu.Lock()
    records := p.records[toolName]
    p.mu.Unlock()

    if len(records) == 0 {
        return &intelligence.ToolProfile{ToolName: toolName}, nil
    }

    profile := &intelligence.ToolProfile{
        ToolName:   toolName,
        TotalCalls: len(records),
    }

    successes := 0
    var totalDuration time.Duration
    durations := make([]time.Duration, 0, len(records))
    totalTokens := 0

    for _, r := range records {
        if r.Success {
            successes++
        }
        totalDuration += r.Duration
        durations = append(durations, r.Duration)
        totalTokens += r.Tokens
        if r.Timestamp.After(profile.LastUsed) {
            profile.LastUsed = r.Timestamp
        }
    }

    profile.SuccessRate = float64(successes) / float64(len(records))
    profile.AvgDuration = totalDuration / time.Duration(len(records))
    profile.AvgTokens = totalTokens / len(records)

    sort.Slice(durations, func(i, j int) bool { return durations[i] < durations[j] })
    p95Idx := int(float64(len(durations)) * 0.95)
    if p95Idx >= len(durations) {
        p95Idx = len(durations) - 1
    }
    profile.P95Duration = durations[p95Idx]

    return profile, nil
}

func (p *InMemoryProfiler) AllProfiles(ctx context.Context) (map[string]*intelligence.ToolProfile, error) {
    p.mu.Lock()
    names := make([]string, 0, len(p.records))
    for name := range p.records {
        names = append(names, name)
    }
    p.mu.Unlock()

    result := make(map[string]*intelligence.ToolProfile)
    for _, name := range names {
        profile, _ := p.Profile(ctx, name)
        result[name] = profile
    }
    return result, nil
}
```

```go
package optimize

import (
    "context"
    "sync"
)

type HistorySelector struct {
    mu      sync.Mutex
    success map[string]int
    total   map[string]int
}

func NewHistorySelector() *HistorySelector {
    return &HistorySelector{
        success: make(map[string]int),
        total:   make(map[string]int),
    }
}

func (s *HistorySelector) Select(ctx context.Context, task string, candidates []string) (string, error) {
    s.mu.Lock()
    defer s.mu.Unlock()

    if len(candidates) == 0 {
        return "", nil
    }

    // 选历史成功率最高的
    bestChoice := candidates[0]
    bestRate := -1.0

    for _, c := range candidates {
        total := s.total[c]
        if total == 0 {
            continue
        }
        rate := float64(s.success[c]) / float64(total)
        if rate > bestRate {
            bestRate = rate
            bestChoice = c
        }
    }
    return bestChoice, nil
}

func (s *HistorySelector) RecordOutcome(ctx context.Context, toolName string, success bool) error {
    s.mu.Lock()
    defer s.mu.Unlock()
    s.total[toolName]++
    if success {
        s.success[toolName]++
    }
    return nil
}
```

```go
// internal/tools/intelligence/create/detector.go
package create

import (
    "context"
    "strings"
    "sync"
    "time"
    "github.com/Gleamseekers/AgentPrimordia/agentprimordia/internal/tools/intelligence"
)

type TraceGapDetector struct {
    mu sync.Mutex
}

func NewTraceGapDetector() *TraceGapDetector {
    return &TraceGapDetector{}
}

func (d *TraceGapDetector) Detect(ctx context.Context, trace []intelligence.ToolCallRecord) ([]intelligence.GapCandidate, error) {
    d.mu.Lock()
    defer d.mu.Unlock()

    // 统计失败模式
    failCounts := make(map[string]int)
    firstSeen := make(map[string]time.Time)
    lastSeen := make(map[string]time.Time)
    sampleErrors := make(map[string]string)

    for _, r := range trace {
        if r.Success {
            continue
        }
        // 从错误消息中提取缺口类型
        key := extractGapKey(r.Error)
        if key == "" {
            continue
        }
        failCounts[key]++
        if _, ok := firstSeen[key]; !ok {
            firstSeen[key] = r.Timestamp
        }
        lastSeen[key] = r.Timestamp
        sampleErrors[key] = r.Error
    }

    var gaps []intelligence.GapCandidate
    for key, count := range failCounts {
        if count >= 2 { // 至少 2 次失败才报告
            gaps = append(gaps, intelligence.GapCandidate{
                Kind:        "missing_tool",
                Key:         key,
                Count:       count,
                SampleError: sampleErrors[key],
                FirstSeen:   firstSeen[key],
                LastSeen:    lastSeen[key],
            })
        }
    }
    return gaps, nil
}

func extractGapKey(errorMsg string) string {
    if strings.Contains(errorMsg, "no suitable tool") {
        // 提取工具类型关键词
        for _, keyword := range []string{"csv", "json", "xml", "yaml", "image", "audio"} {
            if strings.Contains(errorMsg, keyword) {
                return keyword + "_parser"
            }
        }
        return "unknown_tool"
    }
    return ""
}
```
